Integrate A2UI UI generation into a TypeScript, Python, or Dart [agent](../../concepts/glossary/#agent) server using the [Agent SDKs](../../concepts/sdks/#agent-sdk) to stream validated interactive [surfaces](../../concepts/glossary/#surface) to clients and handle multi-turn user actions.

If you are building the client application instead, see [How to Integrate A2UI into a Client](../../guides/client-integration/). To register custom widgets beyond the built-in [Basic catalog](../../specification/v1.0-basic-catalog-implementation-guide/#basic_catalog.json), see [How to Define a Catalog](../../guides/defining-a-catalog/), and for wire vs. compact DSL trade-offs see [Swappable Inference Formats](../../concepts/inference-formats/). Swift and Kotlin target client-side rendering rather than server-side agent generation (see [SDKs](../../concepts/sdks/#agent-sdk)).

## Step 2: Initialize A2uiGenerator

The steps below use a sample e-commerce order assistant (`OrderAgentService` in `order-agent.ts` on TypeScript, `order_agent.py` on Python, and `lib/order_agent.dart` on Dart) to keep the snippets concrete, though your agent can serve any domain. Create one [`A2uiGenerator`](../../concepts/glossary/#a2uigenerator) when your server starts inside that service, wired at the server entry point in [Step 5](#step-5-wire-everything-at-the-server-entry-point).

Register the built-in `v1.0` [Basic catalog](../../specification/v1.0-basic-catalog-implementation-guide/#basic_catalog.json) inside a `CatalogConfig` wrapper and keep a per-session `RendererState` map so follow-up turns can update existing surfaces in place. By default, `A2uiGenerator` uses `ExpressFormatFactory`, which instructs the language model to write compact `<a2ui>` [inference-format](../../concepts/glossary/#inference-format) blocks and compiles them into validated A2UI JSON messages. To load domain-specific `catalog.json` schemas alongside the Basic catalog, see [How to Define a Catalog](../../guides/defining-a-catalog/).

## Step 3: Build the system prompt and stream A2UI messages

Add `prepareTurn` (`prepare_turn` in Python) and `streamTurn` (`stream_turn` in Python) to `OrderAgentService`. On each turn, look up or create the session's `RendererState`, create an `A2uiRequestProcessor` bound to that state, and append `processor.promptSnippet` (`processor.prompt_snippet` in Python) to your agent's role instructions. If each chat turn creates an independent surface instead of updating earlier ones, omit `state` so every response validates against an empty `RendererState`.

Pass incoming LLM token chunks into `processor.openStream()` (`processor.open_stream()` in Python). The stream reader heals progressive string properties such as `text` as tokens arrive, applies each emitted message to `RendererState`, and yields [`createSurface`](../../concepts/messages/#createsurface), [`updateComponents`](../../concepts/messages/#updatecomponents), [`updateDataModel`](../../concepts/messages/#updatedatamodel), and [`deleteSurface`](../../concepts/messages/#deletesurface) messages as soon as they are ready:

<Tabs syncKey="sdk-lang">
  <TabItem label="TypeScript">

```typescript
prepareTurn(sessionId: string): {processor: A2uiRequestProcessor; systemPrompt: string} {
  let state = this.sessions.get(sessionId);
  if (!state) {
    state = new RendererState([basicCatalog]);
    this.sessions.set(sessionId, state);
  }

  const processor = this.generator.createProcessor(
    {'1.0': {supportedCatalogIds: [basicCatalog.id]}},
    undefined,
    {state},
  );
  const systemPrompt = [
    'You are an order assistant. Show order details in an interactive card.',
    processor.promptSnippet,
  ].join('\n\n');

  return {processor, systemPrompt};
}

async *streamTurn(
  processor: A2uiRequestProcessor,
  tokenChunks: AsyncIterable<string>,
): AsyncGenerator<AgentToRendererMessage> {
  const stream = processor.openStream();
  for await (const chunk of tokenChunks) {
    for (const part of stream.feed(chunk)) {
      if (part.type === 'a2ui') {
        for (const message of part.a2ui) {
          yield message;
        }
      }
    }
  }
  for (const part of stream.close()) {
    if (part.type === 'a2ui') {
      for (const message of part.a2ui) {
        yield message;
      }
    }
  }
}
```

```python
def prepare_turn(
    self, session_id: str
) -> tuple[A2uiRequestProcessor, str]:
  state = self.sessions.get(session_id)
  if state is None:
    state = RendererState([self.basic])
    self.sessions[session_id] = state

  processor = self.generator.create_processor(
      {"1.0": {"supportedCatalogIds": [self.basic.catalog_id]}},
      state=state,
  )
  system_prompt = (
      "You are an order assistant. Show order details in an interactive card.\n\n"
      f"{processor.prompt_snippet}"
  )
  return processor, system_prompt


async def stream_turn(
    self,
    processor: A2uiRequestProcessor,
    token_chunks: AsyncIterable[str],
) -> AsyncIterable[dict[str, Any]]:
  stream = processor.open_stream()
  async for chunk in token_chunks:
    for part in stream.feed(chunk):
      if isinstance(part, A2uiPart):
        for message in to_message_dicts(part.a2ui):
          yield message
  for part in stream.close():
    if isinstance(part, A2uiPart):
      for message in to_message_dicts(part.a2ui):
        yield message
```

```dart
({A2uiRequestProcessor processor, String systemPrompt}) prepareTurn(
  String sessionId,
) {
  final state = _sessions.putIfAbsent(
    sessionId,
    () => RendererState([basic]),
  );

  final processor = generator.createProcessor(
    A2uiRendererCapabilities.forCatalogIds(
      [basic.id],
      version: A2uiProtocolVersion.v1_0,
    ),
    state: state,
  );
  final systemPrompt =
      'You are an order assistant. Show order details in an interactive card.\n\n'
      '${processor.promptSnippet}';

  return (processor: processor, systemPrompt: systemPrompt);
}

Stream<Map<String, Object?>> streamTurn(
  A2uiRequestProcessor processor,
  Stream<String> tokenChunks,
) async* {
  final stream = processor.openStream();
  await for (final chunk in tokenChunks) {
    for (final part in stream.feed(chunk)) {
      if (part is A2uiPart) {
        for (final message in part.a2ui) {
          yield message.toJson();
        }
      }
    }
  }
  for (final part in stream.close()) {
    if (part is A2uiPart) {
      for (final message in part.a2ui) {
        yield message.toJson();
      }
    }
  }
}
```

With `ExpressFormatFactory`, the model writes a compact `<a2ui>` block using UAX #31 component identifiers (`confirm_btn`, `confirm_label`) and `$/...` data-model bindings:

```text
<a2ui>
surface("order_summary")
$/orderId = "ORD-4821"
$/summaryText = "2 items ready for pickup"
root = Card(col)
col = Column([headline, confirm_btn])
headline = Text($/summaryText)
confirm_btn = Button(confirm_label, _, Event("confirm_order", {orderId: $/orderId}))
confirm_label = Text("Confirm order")
</a2ui>
```

The parser compiles that block into validated JSON messages (`"version": "1.0"` with `{"@path": "..."}` bindings in `v1.0`, or `"version": "v0.9"` with `{"path": "..."}` bindings in `v0.9.1`). ## Step 4: Handle user actions

When a user interacts with a component such as **Confirm order**, the client renderer resolves the action's `context` bindings against its local `DataModel` and sends an [`action`](../../concepts/messages/#action) message back to your server. Add `handleAction` (`handle_action` in Python) to `OrderAgentService` to convert that action envelope into the next turn's prompt so the model can update `"order_summary"` in place:

<Tabs syncKey="sdk-lang">
  <TabItem label="TypeScript">

```typescript
handleAction(envelope: Record<string, unknown>): string {
  const act = envelope.action as
    | {name: string; surfaceId: string; context: Record<string, unknown>}
    | undefined;
  if (!act) {
    throw new Error('Expected an action message');
  }
  return (
    `User triggered action '${act.name}' on surface '${act.surfaceId}' ` +
    `with context ${JSON.stringify(act.context)}.`
  );
}
```

  </TabItem>
  <TabItem label="Python">

## Step 5: Wire everything at the server entry point

Connect `OrderAgentService` at your server entry point (`server.ts` on TypeScript, `server.py` on Python, and `bin/server.dart` on Dart) behind an HTTP endpoint (`POST /api/agent`) that accepts initial user prompts (`{sessionId, prompt}`) and outbound client action envelopes (`{sessionId, message}`) and streams validated A2UI messages over Server-Sent Events (`text/event-stream`). On follow-up turns, the model passes `update=true` in `surface("order_summary", update=true)` to update the surface already tracked in `RendererState`:

<Tabs syncKey="sdk-lang">
  <TabItem label="TypeScript">

```typescript
import {createServer, type IncomingMessage, type ServerResponse} from 'node:http';
import {OrderAgentService} from './order-agent.js';

interface AgentRequestBody {
  sessionId?: string;
  prompt?: string;
  message?: Record<string, unknown>;
}

const agentService = new OrderAgentService();

async function readJson(req: IncomingMessage): Promise<AgentRequestBody> {
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk));
  }
  return JSON.parse(Buffer.concat(chunks).toString('utf8')) as AgentRequestBody;
}

async function* callModelStream(systemPrompt: string, userPrompt: string): AsyncGenerator<string> {
  // Replace with your LLM SDK streaming call (e.g., Gemini, OpenAI, Anthropic).
  if (userPrompt.startsWith('User triggered action')) {
    yield '<a2ui>\nsurface("order_summary", update=true)\n';
    yield '$/summaryText = "Order ORD-4821 confirmed for pickup"\n</a2ui>';
    return;
  }
  yield '<a2ui>\nsurface("order_summary")\n';
  yield '$/orderId = "ORD-4821"\n$/summaryText = "2 items ready for pickup"\n';
  yield 'root = Card(col)\ncol = Column([headline, confirm_btn])\n';
  yield 'headline = Text($/summaryText)\n';
  yield 'confirm_btn = Button(confirm_label, _, Event("confirm_order", {orderId: $/orderId}))\n';
  yield 'confirm_label = Text("Confirm order")\n</a2ui>';
}

createServer(async (req: IncomingMessage, res: ServerResponse) => {
  if (req.method !== 'POST' || req.url !== '/api/agent') {
    res.writeHead(404).end();
    return;
  }

  const body = await readJson(req);
  const sessionId = body.sessionId ?? 'default';
  const turnPrompt = body.message
    ? agentService.handleAction(body.message)
    : (body.prompt ?? 'Show my order summary');

  const {processor, systemPrompt} = agentService.prepareTurn(sessionId);

  res.writeHead(200, {
    'Content-Type': 'text/event-stream',
    'Cache-Control': 'no-cache',
    Connection: 'keep-alive',
  });

  const tokens = callModelStream(systemPrompt, turnPrompt);
  for await (const msg of agentService.streamTurn(processor, tokens)) {
    res.write(`data: ${JSON.stringify(msg)}\n\n`);
  }
  res.end();
}).listen(8080);
```

## Additional steps

### Negotiate client capabilities

If your team owns both the agent server and the client app, targeting your known catalogs directly as in [Step 3](#step-3-build-the-system-prompt-and-stream-a2ui-messages) avoids runtime capability plumbing. When your agent serves external clients or multiple shipped app versions with different catalog sets, register each supported catalog on `A2uiGenerator`, read `metadata.a2uiRendererCapabilities` (or `metadata.a2uiClientCapabilities` for `v0.9.1` clients) on every request, and check `probe.acceptsState(state)` (`probe.accepts_state(state)` in Python) in case the caller's negotiated [catalogs](../../concepts/catalogs/#catalog-negotiation-and-multi-catalog-surfaces) changed. In TypeScript and Python, all catalogs on one `A2uiGenerator` must target the same protocol version, so keep a separate `v0.9` `A2uiGenerator` when serving both `v1.0` and `v0.9.1` clients (Dart's `A2uiGenerator` groups mixed-version catalogs automatically):

<Tabs syncKey="sdk-lang">
  <TabItem label="TypeScript">

```typescript
import type {RendererCapabilities} from '@a2ui/agent';

prepareNegotiatedTurn(
  sessionId: string,
  capabilitiesPayload?: Record<string, RendererCapabilities>,
): {processor: A2uiRequestProcessor; state: RendererState; systemPrompt: string} {
  const capabilities: Record<string, RendererCapabilities> = capabilitiesPayload ?? {
    '1.0': {supportedCatalogIds: [basicCatalog.id]},
  };

  const probe = this.generator.createProcessor(capabilities);
  let state = this.sessions.get(sessionId);
  if (!state || !probe.acceptsState(state)) {
    state = probe.createState();
    this.sessions.set(sessionId, state);
  }

  const processor = this.generator.createProcessor(capabilities, undefined, {state});
```

### Synchronize client data models and handle function RPCs

Beyond user `action` messages, clients can send synchronized surface [data models](../../concepts/glossary/#data-model), validation error reports, and `v1.0` remote function calls:

1. When a surface is created with [`sendDataModel: true`](../../concepts/data-binding/#senddatamodel-true) (`surface("order_summary", sendDataModel=true)` in Express), local two-way input edits attach the surface snapshot under `metadata.a2uiRendererDataModel.surfaces[surfaceId]` (`metadata.a2uiClientDataModel.surfaces[surfaceId]` in `v0.9.1`) alongside outbound actions and prompts.
2. [`callAgentFunction`](../../concepts/messages/#callagentfunction-v10-only) (`v1.0` only) invokes a catalog function hosted on the agent and expects a correlated [`agentFunctionResponse`](../../concepts/messages/#agentfunctionresponse-v10-only) without running an LLM turn. In Python, register handlers on your catalog with `AgentFunction` and `catalog.copy_with(agent_functions=[...])` and dispatch incoming envelopes through `RendererToAgentMessageProcessor` (`from a2ui.rpc import RendererToAgentMessageProcessor`); in TypeScript and Dart, inspect `envelope.callAgentFunction` directly.
3. [`error`](../../specification/v1.0/schemas/#renderer_to_agent.json/properties/error) reports a client-side `VALIDATION_FAILED` or function failure (`code`, `surfaceId`, `message`) that you can forward into a repair prompt.

### Validate whole responses and customize inference formats

When you prefer atomic validation over incremental token streaming, call `processor.parseResponse(rawText)` (`processor.parse_response(raw_text)` in Python). If the model invents an unknown component, omits `root`, or leaves a dangling child ID, `RendererState` rolls back every change from that turn and throws `A2uiError`, whose message you can feed back to the model for a self-correcting retry.

You can also customize prompt size and output syntax on `CatalogConfig` and `A2uiGenerator`:

1. Prune large catalogs to the exact components and functions your prompt needs with `ComponentPruningTransformer` and `FunctionPruningTransformer`.
2. Tune `ExpressFormatFactory` options (`allowedMessages` / `allowed_messages`, `progressiveKeys` / `progressive_keys`, and `surfaceId` / `surface_id` in TypeScript and Python), or pass `DirectJsonFormatFactory` if your model or fine-tuned checkpoint writes raw JSON envelopes inside `<a2ui-json>` tags (see [Swappable Inference Formats](../../concepts/inference-formats/)).
3. In single-process TypeScript or Dart apps that run both the agent and a local `MessageProcessor`, wrap the request processor in `StreamedAnswer` (`new StreamedAnswer(processor, {renderer})` in TypeScript or `StreamedAnswer(processor, renderer: renderer)` in Dart), which binds `RendererState.of(renderer)` directly without JSON serialization.
4. In Python, compose reusable server-side component templates with `@macro` and expand them into Basic catalog primitives via [`MacroExpander`](../../concepts/glossary/#macroexpander) (`from a2ui.transformers.macros import MacroExpander, macro`) so mobile and web clients render new composite layouts without a client release.

```python
    before_agent_callback=a2ui_rpc_callback(catalogs=active_catalogs),
)

executor_config = A2aAgentExecutorConfig(
    event_converter=A2uiEventConverter(a2ui_processor=processors.get),
)

# In your request executor before running the session:
# processors.set(session.id, processor if a2ui_active else None)
```

<Aside type="caution" title="Avoid raw prompt strings in LlmAgent.instruction">
  ADK substitutes `{name}` placeholders in plain string instructions from session state.
  Concatenating `processor.prompt_snippet` into a raw `LlmAgent.instruction` string raises
  `KeyError` on A2UI braces (such as `${expression}` or `\p{XID_Start}`). Always wrap your
  instruction in `A2uiInstruction`, which resolves your own `{store_city}` template variables while
  leaving the A2UI prompt untouched.
</Aside>

### A2A protocol

Install `a2ui-a2a` to advertise A2UI support on your `AgentCard`, negotiate the [A2A Extension Specification](../../specification/v1.0-a2ui-extension-specification/) per request, wrap validated messages as `application/a2ui+json` `DataPart` payloads (`to_a2a_parts` and `stream_response_to_parts`), and extract or reply to inbound A2UI messages (`get_a2ui_messages` and `create_a2ui_parts`):

```bash
pip install a2ui-a2a
```

```python
from a2a.types import AgentCapabilities, AgentCard

from a2ui.a2a import (
    create_a2ui_parts,
    get_a2ui_agent_extension,
    get_a2ui_messages,
    stream_response_to_parts,
    try_activate_a2ui_extension,
)
from a2ui.core.basic_catalog import BasicCatalog
from a2ui.core.schema import ProtocolVersion
from a2ui.rpc import RendererToAgentMessageProcessor

basic = BasicCatalog(ProtocolVersion.V1_0)

agent_card = AgentCard(
    name="Order Agent",
    description="Streams interactive order summaries with A2UI.",
    url="http://localhost:10002/",
    version="1.0.0",
    default_input_modes=["text/plain"],
    default_output_modes=["text/plain"],
    capabilities=AgentCapabilities(
        streaming=True,
        extensions=[
            get_a2ui_agent_extension(
                ProtocolVersion.V1_0,
                supported_catalog_ids=[basic.catalog_id],
            ),
        ],
    ),
    skills=[],
)

# Inside your A2A AgentExecutor.execute(context, event_queue):
activated_version = try_activate_a2ui_extension(context, agent_card)
if activated_version is not None:
  rpc_processor = RendererToAgentMessageProcessor([basic])
  inbound = get_a2ui_messages(context.message)
  results = [await rpc_processor.process_message(msg) for msg in inbound]
  rpc_reply_parts = create_a2ui_parts(
      reply for result in results for reply in result.replies
  )
  if rpc_reply_parts:
    await send_parts(event_queue, rpc_reply_parts)
  else:
    async for a2a_part in stream_response_to_parts(
        processor.open_stream(), token_stream, activated_version
    ):
      await send_parts(event_queue, [a2a_part])
```

### AG-UI and CopilotKit

To stream A2UI surfaces over [AG-UI](https://docs.ag-ui.com/), enable the `a2ui` option on `CopilotRuntime` in `@copilotkit/runtime`. CopilotKit injects the A2UI generation tool and translates tool calls and state deltas into AG-UI events across Python and TypeScript backends, including Google ADK (`ag_ui_adk`), LangGraph (`CopilotKitMiddleware`), and AWS Strands (`@ag-ui/aws-strands`). See the [CopilotKit A2UI documentation](https://docs.copilotkit.ai/generative-ui/a2ui) for runtime setup details:

```typescript
import {CopilotRuntime} from '@copilotkit/runtime';

const runtime = new CopilotRuntime({
  agents: {
    order_agent: myOrderAgent,
  },
  a2ui: {
    injectA2UITool: true, // Optional (default: false)
  },
});
```

For Model Context Protocol tool servers that attach interactive UI resources (`ui://`), see [A2UI over MCP](../../guides/a2ui-over-mcp/) and [MCP Integration Overview](../../guides/mcp-overview/).
